cmake_minimum_required(VERSION 3.22)
project(WhatsFeather VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Quick QuickControls2 QuickDialogs2 Network Multimedia Test)
qt_standard_project_setup(REQUIRES 6.5)
qt_add_library(feather-core STATIC desktop/src/communityfoldermodel.cpp desktop/src/communityfoldermodel.h desktop/src/controller.cpp desktop/src/controller_extensions.cpp desktop/src/controller.h desktop/src/jsonmodel.cpp desktop/src/jsonmodel.h desktop/src/localexpressionstore.cpp desktop/src/localexpressionstore.h desktop/src/voicenoterecorder.cpp desktop/src/voicenoterecorder.h)
target_sources(feather-core PRIVATE desktop/src/desktopnotifications.cpp desktop/src/desktopnotifications.h desktop/src/controller_notifications.cpp desktop/src/controller_expressions.cpp)
target_link_libraries(feather-core PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::Multimedia)
target_include_directories(feather-core PUBLIC desktop/src)
qt_add_executable(whatsfeather desktop/src/main.cpp desktop/src/polishtest.cpp desktop/src/communitytest.cpp desktop/src/communitycachetest.cpp desktop/src/mediadownloadtest.cpp desktop/src/chatrostertest.cpp desktop/src/contactcardstest.cpp desktop/src/locationtest.cpp desktop/src/polltest.cpp desktop/src/listmessagetest.cpp desktop/src/buttonstest.cpp desktop/src/textcardtest.cpp)
if(Qt6_VERSION VERSION_GREATER_EQUAL "6.8")
  qt_policy(SET QTP0004 NEW)
endif()
set_source_files_properties(desktop/qml/Main.qml PROPERTIES QT_RESOURCE_ALIAS Main.qml)
set_source_files_properties(desktop/qml/IconButton.qml PROPERTIES QT_RESOURCE_ALIAS IconButton.qml)
set(FEATHER_QML Main IconButton FloatingComposer ExpressionPicker MessageBubble MediaViewer AttachmentDialog GifBrowser ActionButton ChatWallpaper CommunityFolderRow NotificationPreferences CachedConversationInfo SharedMediaContent)
set(FEATHER_QML_FILES)
foreach(component IN LISTS FEATHER_QML)
 set_source_files_properties(desktop/qml/${component}.qml PROPERTIES QT_RESOURCE_ALIAS ${component}.qml)
 list(APPEND FEATHER_QML_FILES desktop/qml/${component}.qml)
endforeach()
qt_add_qml_module(whatsfeather URI WhatsFeather VERSION 1.0 QML_FILES ${FEATHER_QML_FILES})
qt_add_resources(whatsfeather "icons" PREFIX "/icons" BASE "desktop/icons" FILES desktop/icons/search.svg desktop/icons/send.svg desktop/icons/settings.svg desktop/icons/back.svg desktop/icons/phone.svg desktop/icons/video.svg desktop/icons/feather.svg desktop/icons/attach.svg desktop/icons/emoji.svg desktop/icons/more.svg desktop/icons/chevron.svg desktop/icons/close.svg desktop/icons/chat.svg desktop/icons/file.svg desktop/icons/mic.svg desktop/icons/stop.svg)
set(FEATHER_APP_ICON_SIZES 16 24 32 48 64 128 256 512)
set(FEATHER_APP_ICON_FILES desktop/assets/app-icons/whatsfeather.png)
foreach(size IN LISTS FEATHER_APP_ICON_SIZES)
 list(APPEND FEATHER_APP_ICON_FILES desktop/assets/app-icons/whatsfeather-${size}.png)
endforeach()
qt_add_resources(whatsfeather "application-icons" PREFIX "/app-icons" BASE "desktop/assets/app-icons" FILES ${FEATHER_APP_ICON_FILES})
target_link_libraries(whatsfeather PRIVATE feather-core Qt6::Quick Qt6::QuickControls2 Qt6::QuickDialogs2 Qt6::Multimedia)
qt_add_resources(whatsfeather "chat-backgrounds" PREFIX "/backgrounds" BASE "desktop/assets/backgrounds" FILES desktop/assets/backgrounds/chat-dark.webp desktop/assets/backgrounds/chat-green.webp)
include(GNUInstallDirs)
install(TARGETS whatsfeather RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS "${CMAKE_BINARY_DIR}/bin/whatsfeatherd" DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES "${CMAKE_SOURCE_DIR}/LICENSE" "${CMAKE_SOURCE_DIR}/LICENSES/ASSET-LICENSES.md" DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/WhatsFeather)
install(FILES desktop/packaging/whatsfeather.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
foreach(size IN LISTS FEATHER_APP_ICON_SIZES)
 install(FILES desktop/assets/app-icons/whatsfeather-${size}.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/${size}x${size}/apps RENAME whatsfeather.png)
endforeach()
enable_testing()
qt_add_executable(feather-notification-tests desktop/tests/test_desktopnotifications.cpp)
target_link_libraries(feather-notification-tests PRIVATE feather-core Qt6::Test)
add_test(NAME desktop-notifications COMMAND feather-notification-tests)
set_tests_properties(desktop-notifications PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
qt_add_executable(feather-community-tests desktop/tests/test_communityfolders.cpp)
target_link_libraries(feather-community-tests PRIVATE feather-core Qt6::Test)
add_test(NAME community-folders COMMAND feather-community-tests)
qt_add_executable(feather-tests desktop/tests/test_core.cpp)
target_link_libraries(feather-tests PRIVATE feather-core Qt6::Test)
add_test(NAME native-core COMMAND feather-tests)
qt_add_executable(feather-audio-tests desktop/tests/test_voicenoterecorder.cpp)
target_link_libraries(feather-audio-tests PRIVATE feather-core Qt6::Test)
add_test(NAME voice-note-recorder COMMAND feather-audio-tests)
set_tests_properties(native-core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
set_tests_properties(voice-note-recorder PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME text-card-source COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_text_card.py)
add_test(NAME text-card-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/test_text_cards.py)
